// Inicializar Supabase (já deve estar no seu HTML)
const supabaseClient = window.supabase.createClient(SUPABASE_URL, SUPABASE_KEY);

async function loadCollectorStatus() {
    try {
        // Buscar estatísticas dos logs
        const { data: logs, error: logsError } = await supabaseClient
            .from('collection_logs')
            .select('*')
            .order('created_at', { ascending: false })
            .limit(1)
            .single();

        if (logsError) throw logsError;

        // Buscar total de partidas
        const { count: totalCount, error: countError } = await supabaseClient
            .from('matches')
            .select('*', { count: 'exact', head: true });

        // Atualizar cards
        document.getElementById('total-matches').textContent = totalCount || '0';
        document.getElementById('new-matches').textContent = logs?.new_matches || '0';
        document.getElementById('last-run').textContent = logs?.created_at 
            ? new Date(logs.created_at).toLocaleString('pt-BR') 
            : '-';
        
        // Próxima coleta em 30 minutos
        const nextRun = new Date(new Date(logs?.created_at).getTime() + 30 * 60 * 1000);
        document.getElementById('next-run').textContent = nextRun.toLocaleString('pt-BR');

    } catch (error) {
        console.error('Erro ao carregar status:', error);
        document.getElementById('collector-status-text').textContent = 'Erro na conexão';
    }
}

async function loadLatestMatches() {
    try {
        const { data: matches, error } = await supabaseClient
            .from('matches')
            .select('*, ligas(name)')
            .order('match_time', { ascending: false })
            .limit(20);

        if (error) throw error;

        const tbody = document.getElementById('latest-matches-tbody');
        
        if (!matches || matches.length === 0) {
            tbody.innerHTML = `
                <tr>
                    <td colspan="7" style="text-align: center; padding: 40px; color: var(--color-text-secondary);">
                        Nenhuma partida coletada ainda
                    </td>
                </tr>
            `;
            return;
        }

        tbody.innerHTML = matches.map(match => `
            <tr>
                <td>${new Date(match.match_time).toLocaleString('pt-BR')}</td>
                <td><span style="color: var(--color-primary); font-weight: 600;">${match.ligas?.name || 'N/A'}</span></td>
                <td>${match.home_team}</td>
                <td style="font-weight: 700; color: var(--color-warning);">${match.full_time_home} - ${match.full_time_away}</td>
                <td>${match.away_team}</td>
                <td>
                    <span class="result-badge ${match.result === 'Casa' ? 'win' : match.result === 'Fora' ? 'loss' : 'draw'}">
                        ${match.result}
                    </span>
                </td>
                <td>${match.half_time_home} - ${match.half_time_away}</td>
            </tr>
        `).join('');

    } catch (error) {
        console.error('Erro ao carregar partidas:', error);
    }
}

// Função para trocar de aba (certifique-se que existe no seu código)
function switchTab(tabName) {
    // Esconder todos os tabs
    const allContents = document.querySelectorAll('.tab-content');
    allContents.forEach(content => content.style.display = 'none');

    // Mostrar o selecionado
    const selectedContent = document.getElementById(tabName + '-content');
    if (selectedContent) {
        selectedContent.style.display = 'block';
        
        // Se for collector, carregar dados
        if (tabName === 'collector') {
            loadCollectorStatus();
            loadLatestMatches();
        }
    }

    // Atualizar nav items ativos
    document.querySelectorAll('.nav-item').forEach(item => {
        item.classList.remove('active');
    });
    event.target.closest('.nav-item')?.classList.add('active');
}

// Atualizar a cada 30 segundos para refletir coletas em tempo real
setInterval(() => {
    if (document.getElementById('collector-content').style.display !== 'none') {
        loadCollectorStatus();
        loadLatestMatches();
    }
}, 30 * 1000);
